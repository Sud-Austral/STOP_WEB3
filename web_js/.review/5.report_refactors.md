```javascript
// index_vistas2.html

App.loadView = async (viewId) => {
    if (currentFetchController) currentFetchController.abort();
    currentFetchController = new AbortController();
    try {
        const response = await fetch(url, { signal: currentFetchController.signal });
        // ...
    } catch (e) {
        if (e.name === 'AbortError') return; // Ignorar cancelación intencional
        console.error("Error loading view:", e);
    }
};

// Centralizar waitForData con timeout
App.waitForData = (timeoutMs = 10000) => {
    return new Promise((resolve, reject) => {
        const start = Date.now();
        const check = () => {
             if (window.STATE_DATA?.isLoaded) resolve();
             else if (Date.now() - start > timeoutMs) reject("Timeout data load");
             else setTimeout(check, 100);
        };
        check();
    });
};
```
